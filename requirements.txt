# ------------------------------------------------------------------------------
# 1. HOST SYSTEM PREREQUISITES (Must be installed manually on a fresh machine)
# ------------------------------------------------------------------------------

# Java Development Kit (MANDATORY)
# Version: OpenJDK 21 (LTS) or Eclipse Temurin 21 (LTS)
# Environment: JAVA_HOME must point to JDK 21 installation directory
# Verification: java -version  (must report version 21.x.x)
JDK=21

# Container Engine (RECOMMENDED for Database and Testcontainers)
# Tool: Docker Desktop (Windows / macOS) or Docker Engine + Compose (Linux)
# Version: Docker Engine 24.0+ / Docker Compose v2+
# Purpose: Runs PostgreSQL 15 database container and automated integration tests
# Verification: docker --version && docker compose version
DOCKER=24.0+

# Alternative Database (If Docker is NOT used)
# Tool: PostgreSQL Server
# Version: 15.0 or higher (PostgreSQL 15, 16, 17, 18 supported)
# Required Extension: btree_gist (included with standard PostgreSQL distribution)
# Verification: psql --version
POSTGRESQL=15+

# Version Control
# Tool: Git
# Verification: git --version
GIT=2.30+


# ------------------------------------------------------------------------------
# 2. LOCAL CONFIGURATION SETUP (Must be created on a fresh machine)
# ------------------------------------------------------------------------------
# 
# 1. Copy the example configuration template:
#    cp config/application.properties.example config/application.properties
#    (or copy .env.example to .env)
#
# 2. Configure your database connection in config/application.properties:
#    db.url=jdbc:postgresql://localhost:5432/dras
#    db.username=dras_app
#    db.password=<your_password>
#    db.pool.size=5
#    db.timeout.ms=10000
